```python
# 高德地图MCP服务封装
class AmapService:
    def __init__(self):
        self.api_key = settings.AMAP_API_KEY
        self.base_url = 'https://restapi.amap.com/v3'
    
    def search_nearby_pois(self, location, keywords, radius=100000):
        """搜索周边POI"""
        params = {
            'key': self.api_key,
            'location': f"{location['longitude']},{location['latitude']}",
            'keywords': keywords,
            'radius': radius,
            'output': 'json'
        }
        
        response = requests.get(f"{self.base_url}/place/around", params=params)
        return self.parse_poi_response(response.json())
    
    def parse_poi_response(self, data):
        """解析POI响应数据"""
        hospitals = []
        for poi in data.get('pois', []):
            hospital = {
                'id': poi['id'],
                'name': poi['name'],
                'address': poi['address'],
                'location': {
                    'longitude': float(poi['location'].split(',')[0]),
                    'latitude': float(poi['location'].split(',')[1])
                },
                'distance': int(poi['distance']),
                'tel': poi.get('tel', '')
            }
            hospitals.append(hospital)
        return hospitals
```

```python
# 症状分析服务
class SymptomAnalysisService:
    def __init__(self):
        self.symptom_keywords = self.load_symptom_keywords()
    
    def analyze_symptoms(self, symptom_text):
        """分析症状文本"""
        # 1. 关键词提取
        keywords = self.extract_keywords(symptom_text)
        
        # 2. 调用Web搜索MCP获取相关疾病
        related_diseases = self.web_search_service.search_medical(keywords)
        
        # 3. 匹配对应科室
        matched_departments = self.match_departments(related_diseases)
        
        return {
            'keywords': keywords,
            'related_diseases': related_diseases,
            'recommended_departments': matched_departments
        }
    
    def extract_keywords(self, text):
        """提取症状关键词"""
        # 使用jieba分词
        import jieba.analyse
        keywords = jieba.analyse.extract_tags(text, topK=10)
        return keywords
```

```python
# 医生推荐服务
class DoctorRecommendationService:
    def recommend_doctors(self, user_id, symptoms, location, filters=None):
        """推荐医生"""
        # 1. 症状分析
        symptom_analysis = self.symptom_service.analyze_symptoms(symptoms)
        
        # 2. 获取候选医生
        candidate_doctors = self.get_candidate_doctors(
            symptom_analysis['recommended_departments']
        )
        
        # 3. 计算推荐分数
        scored_doctors = []
        for doctor in candidate_doctors:
            score = self.calculate_doctor_score(doctor, location, symptom_analysis)
            scored_doctors.append({
                'doctor': doctor,
                'score': score,
                'match_reason': self.get_match_reason(doctor, symptom_analysis)
            })
        
        # 4. 排序返回
        scored_doctors.sort(key=lambda x: x['score'], reverse=True)
        return scored_doctors[:10]  # 返回前10个推荐
    
    def calculate_doctor_score(self, doctor, location, symptom_analysis):
        """计算医生推荐分数"""
        # 专业匹配度 (40%)
        specialty_score = self.calculate_specialty_match(doctor, symptom_analysis)
        
        # 患者评价 (30%)
        rating_score = doctor.average_rating / 5.0  # 归一化
        
        # 距离因素 (20%)
        distance_score = self.calculate_distance_score(doctor.hospital, location)
        
        # 接诊能力 (10%)
        capacity_score = self.calculate_capacity_score(doctor)
        
        total_score = (
            specialty_score * 0.4 +
            rating_score * 0.3 +
            distance_score * 0.2 +
            capacity_score * 0.1
        )
        
        return total_score
```

```python
# 排班服务
class ScheduleService:
    def get_doctor_schedules(self, doctor_id, start_date, end_date):
        """获取医生排班"""
        # 1. 查询数据库排班
        db_schedules = Schedule.objects.filter(
            doctor_id=doctor_id,
            schedule_date__range=[start_date, end_date],
            is_available=True
        ).order_by('schedule_date', 'start_time')
        
        # 2. 检查是否需要更新排班数据
        if self.need_schedule_update(doctor_id):
            # 调用Web爬虫MCP获取最新排班
            updated_schedules = self.web_crawler_service.get_doctor_schedules(doctor_id)
            self.update_schedule_data(doctor_id, updated_schedules)
            db_schedules = updated_schedules
        
        return self.format_schedules(db_schedules)
    
    def create_appointment(self, user_id, doctor_id, schedule_id, time_slot):
        """创建预约"""
        # 1. 检查时间冲突
        if self.has_time_conflict(user_id, time_slot):
            raise ValidationError('时间冲突，请选择其他时段')
        
        # 2. 检查医生排班可用性
        schedule = Schedule.objects.get(id=schedule_id)
        if schedule.booked_count >= schedule.max_patients:
            raise ValidationError('该时段已满，请选择其他时段')
        
        # 3. 创建预约记录
        appointment = Appointment.objects.create(
            user_id=user_id,
            doctor_id=doctor_id,
            schedule_id=schedule_id,
            appointment_date=schedule.schedule_date,
            appointment_time=time_slot,
            status='pending'
        )
        
        # 4. 更新排班已预约数
        schedule.booked_count += 1
        schedule.save()
        
        return appointment
```

```python
# 路线规划服务
class RoutePlanningService:
    def get_route_planning(self, origin, destination, method='driving'):
        """获取路线规划"""
        # 检查缓存
        cache_key = f"route_{origin}_{destination}_{method}"
        cached_route = cache.get(cache_key)
        if cached_route:
            return cached_route
        
        # 调用高德地图MCP
        amap_service = AmapService()
        route_data = amap_service.route_planning(origin, destination, method)
        
        # 缓存结果（5分钟）
        cache.set(cache_key, route_data, 300)
        
        return route_data
    
    def get_multiple_routes(self, origin, destinations):
        """获取多路线规划（比较不同医院）"""
        routes = {}
        for hospital in destinations:
            route = self.get_route_planning(origin, hospital.location)
            routes[hospital.id] = {
                'distance': route['distance'],
                'duration': route['duration'],
                'route_detail': route
            }
        return routes
```

## 🔄 模块间协作流程

### 预约完整流程
```
用户定位 → 医院发现 → 科室选择 → 医生选择 → 时间选择 → 
预约确认 → 路线规划 → 天气提醒 → 就诊提醒 → 就诊完成
```

### 数据流设计
- **用户请求**：前端收集用户输入和位置信息
- **业务处理**：后端调用相应服务处理业务逻辑
- **MCP集成**：需要外部数据时调用MCP服务
- **数据存储**：处理结果存入数据库
- **响应返回**：格式化数据返回前端
